## ⚡ 性能优化

### 1. 渲染优化

#### React.memo - 避免不必要的重渲染

```typescript
// ❌ 不使用 memo，每次父组件更新都会重渲染
export const DeviceCard = ({ device, onPress }) => {
  return <TouchableOpacity onPress={() => onPress(device)}>...</TouchableOpacity>;
};

// ✅ 使用 memo，只有 props 变化时才重渲染
export const DeviceCard = React.memo(({ device, onPress }) => {
  return <TouchableOpacity onPress={() => onPress(device)}>...</TouchableOpacity>;
});
```

#### useMemo - 缓存计算结果

```typescript
// ❌ 每次渲染都重新计算
const onlineCount = devices.filter(d => d.online).length;
const offlineCount = devices.length - onlineCount;

// ✅ 使用 useMemo 缓存计算结果
const { onlineCount, offlineCount } = useMemo(() => ({
  onlineCount: devices.filter(d => d.online).length,
  offlineCount: devices.length - devices.filter(d => d.online).length,
}), [devices]);
```

#### useCallback - 缓存事件处理函数

```typescript
// ❌ 每次渲染都创建新函数，导致子组件重渲染
const handleDevicePress = (device) => {
  setSelectedDevice(device);
};

// ✅ 使用 useCallback 缓存函数
const handleDevicePress = useCallback((device) => {
  setSelectedDevice(device);
}, []);
```

### 2. 列表优化

#### FlatList - 虚拟化长列表

```typescript
// ❌ 使用 ScrollView，渲染所有项目
<ScrollView>

  initialNumToRender={10}

  windowSize={5}
/>
```

## 🔄 状态管理

### Zustand Store 示例

#### deviceStore.ts

```typescript
interface DeviceState {
  devices: Device[];
  loading: boolean;
  error: string | null;
  
  // Actions
  loadDevices: () => Promise<void>;
  addDevice: (device: Device) => Promise<void>;
  removeDevice: (deviceId: string) => Promise<void>;
  updateDeviceStatus: (deviceId: string, online: boolean) => void;
}

export const useDeviceStore = create<DeviceState>((set, get) => ({
  devices: [],
  loading: false,
  error: null,
  
  loadDevices: async () => {
    set({ loading: true, error: null });
    try {
      const stored = await AsyncStorage.getItem('@devices');
      if (stored) {
        const devices = JSON.parse(stored);
        set({ devices, loading: false });
      }
    } catch (error) {
      set({ error: error.message, loading: false });
    }
  },
  
  addDevice: async (device) => {
    const { devices } = get();
    const updated = [...devices, device];
    set({ devices: updated });
    await AsyncStorage.setItem('@devices', JSON.stringify(updated));
  },
  
  removeDevice: async (deviceId) => {
    const { devices } = get();
    const updated = devices.filter(d => d.id !== deviceId);
    set({ devices: updated });
    await AsyncStorage.setItem('@devices', JSON.stringify(updated));
  },
  
  updateDeviceStatus: (deviceId, online) => {
    const { devices } = get();
    const updated = devices.map(device =>
      device.id === deviceId
        ? { ...device, online, lastSeen: new Date() }
        : device
    );
    set({ devices: updated });
  },
}));
```

## 🎣 Custom Hooks

### useAIChat Hook

```typescript
export const useAIChat = (device: Device, sessionId: string) => {
  const [messages, setMessages] = useState<Message[]>([]);
  const [isProcessing, setIsProcessing] = useState(false);
  
  const sendMessage = useCallback(async (text: string) => {
    setIsProcessing(true);
    
    try {
      // 添加用户消息
      const userMessage: Message = {
        id: Date.now().toString(),
        role: 'user',
        content: text,
        timestamp: new Date(),
      };
      setMessages(prev => [...prev, userMessage]);
      
      // 调用 AI Service
      const aiService = AIProviderFactory.create('openai', apiKey);
      const response = await aiService.sendRequest(text, {
        conversationHistory: messages,
      });
      
      // 添加 AI 响应
      const aiMessage: Message = {
        id: (Date.now() + 1).toString(),
        role: 'assistant',
        content: response.explanation,
        timestamp: new Date(),
        metadata: { command: response.command },
      };
      setMessages(prev => [...prev, aiMessage]);
    } catch (error) {
      console.error('Error sending message:', error);
    } finally {
      setIsProcessing(false);
    }
  }, [device, sessionId, messages]);
  
  return {
    messages,
    isProcessing,
    sendMessage,
  };
};
```

### useSSHConnection Hook

```typescript
export const useSSHConnection = (device: Device) => {
  const [connected, setConnected] = useState(false);
  const [sessionId, setSessionId] = useState<string | null>(null);
  
  const connect = useCallback(async () => {
    try {
      const sshService = getSSHService();
      const session = await sshService.connect(device);
      setSessionId(session.id);
      setConnected(true);
    } catch (error) {
      console.error('SSH connection failed:', error);
      throw error;
    }
  }, [device]);
  
  const disconnect = useCallback(async () => {
    if (sessionId) {
      const sshService = getSSHService();
      await sshService.disconnect(sessionId);
      setSessionId(null);
      setConnected(false);
    }
  }, [sessionId]);
  
  return {
    connected,
    sessionId,
    connect,
    disconnect,
  };
};
```

## 🎯 最佳实践

### 1. 组件设计原则

- **单一职责**: 每个组件只负责一个功能
- **高内聚低耦合**: 组件内部逻辑紧密，组件间依赖少
- **可复用性**: 提取通用组件，避免重复代码
- **可测试性**: 组件易于单元测试

### 2. 性能优化原则

- **避免过度优化**: 先测量，再优化
- **使用 React DevTools**: 分析性能瓶颈
- **懒加载**: 按需加载组件和资源
- **代码分割**: 拆分大文件，减少初始加载时间

### 3. 代码规范

- **TypeScript 严格模式**: 启用所有类型检查
- **ESLint**: 统一代码风格
- **Prettier**: 自动格式化代码
- **注释**: 复杂逻辑添加注释

### 4. 测试策略

- **单元测试**: 测试独立函数和 Hooks
- **组件测试**: 测试组件渲染和交互
- **集成测试**: 测试多个组件协作
- **E2E 测试**: 测试完整用户流程
